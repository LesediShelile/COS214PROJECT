232void WorkflowInstance::finish(WorkflowStatus finalStatus, const std::string& defaultOutcome) {
233 status = finalStatus;
234 if (outcome.empty()) {
235 outcome = defaultOutcome;
236 }
237 // Observer: tell every listener the workflow has ended.
238 WorkflowEvent event(EventType::WORKFLOW_FINISHED, instanceID, currentTimestamp(), outcome);

245void WorkflowInstance::connectPublisher() {
246 // Give every item in the cloned tree this instance's publisher.
247
248 std::unique_ptr<WorkItemIterator> it(rootWorkItem->createIterator());
249 for (it->first(); !it->isDone(); it->next()) {
250 it->current()->setPublisher(publisher);
