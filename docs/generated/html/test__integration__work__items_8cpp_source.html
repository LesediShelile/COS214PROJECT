44std::vector<std::string> availableIds(WorkItem& root) {
45 std::vector<std::string> ids;
46 std::unique_ptr<WorkItemIterator> it(root.createAvailableIterator());
47 for (it->first(); !it->isDone(); it->next()) ids.push_back(it->current()->getItemID());
48 return ids;

53TEST_CASE("Commands run on a stacked decorator and undo restores each state") {
54 EventPublisher pub;
55 Counter counter;
56 pub.attach(&counter);
57 Role role("Nurse", 1);
58 Participant p("p1", "Sam", &role);
59 std::unique_ptr<WorkItem> t(decorated("t1", &pub, ""));
60 t->makeAvailable();
61
62 CommandInvoker inv;
63 inv.executeCommand(new AssignCommand(t.get(), &p));
64 inv.executeCommand(new StartCommand(t.get()));
65 inv.executeCommand(new CompleteCommand(t.get()));
66 CHECK(t->getStateName() == "Completed");
67 CHECK(t->getData("priority") == "5");
68
69 CHECK(inv.undoLast());
70 CHECK(t->getStateName() == "Started");
71 CHECK(inv.undoLast());
72 CHECK(t->getStateName() == "Assigned");
73 CHECK(inv.undoLast());
74 CHECK(t->getStateName() == "Available");
75 CHECK_FALSE(inv.canUndo());
76
77 // makeAvailable + the three commands each published one audit event
78 CHECK(counter.count == 4);
79}

105TEST_CASE("A sequential stage of decorated tasks offers one task at a time") {
106 EventPublisher pub;
107 Role role("Staff", 1);
108 Participant p("p1", "Pat", &role);
109 SequentialStage root("root", "admission");
110 root.add(decorated("a", &pub, ""));
111 root.add(decorated("b", &pub, ""));
112 root.add(decorated("c", &pub, ""));
113
114 root.makeAvailable();
115 CHECK(availableIds(root) == std::vector<std::string>(1, "a"));
116
117 CommandInvoker inv;
118 std::vector<WorkItem*> kids = root.getChildren();
119 for (std::size_t i = 0; i < kids.size(); ++i) {
120 inv.executeCommand(new AssignCommand(kids[i], &p));
121 inv.executeCommand(new StartCommand(kids[i]));
122 inv.executeCommand(new CompleteCommand(kids[i]));
123 root.activateChildren();
124 if (i + 1 < kids.size())
125 CHECK(availableIds(root) == std::vector<std::string>(1, kids[i + 1]->getItemID()));
126 }
127 CHECK(root.isFinished());
128 CHECK(availableIds(root).empty());
129}

139TEST_CASE("A cloned stage tree of decorated tasks runs independently of the original") {
140 EventPublisher pub;
141 Counter counter;
142 pub.attach(&counter);
143 Role role("Staff", 1);
144 Participant p("p1", "Pat", &role);
145
146 SequentialStage* original = new SequentialStage("root", "template");
147 original->add(decorated("a", &pub, ""));
148 original->add(decorated("b", &pub, ""));
149 std::unique_ptr<WorkItem> copy(original->clone());
150
151 copy->makeAvailable();
152 CommandInvoker inv;
153 WorkItem* first = copy->getChildren()[0];
154 inv.executeCommand(new AssignCommand(first, &p));
155 inv.executeCommand(new StartCommand(first));
156 inv.executeCommand(new CompleteCommand(first));
157 CHECK(first->isFinished());
158 CHECK(first->getData("priority") == "5"); // decorator survived the clone
159
160 // the template is untouched, and still behaves like a fresh workflow
161 CHECK_FALSE(original->isFinished());
162 CHECK(original->getChildren()[0]->getStateName() == "Created");
163 original->makeAvailable();
164 CHECK(availableIds(*original) == std::vector<std::string>(1, "a"));
165 delete original;
166}
