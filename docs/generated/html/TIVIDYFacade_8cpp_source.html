50TIVIDYFacade::~TIVIDYFacade() {
51 /* The routers point at instances, so they go first. Instances hold only
52 non-owning pointers to participants, so participants go last.*/
53
54 for (auto& router : routers) {

66 }
67 definitions.clear();
68
69 for (auto& entry : participants) {
70 delete entry.second;
71 }

80void TIVIDYFacade::registerDefinition(WorkflowDefinition* definition) {

90 // A definition must not change once instances can be created from it.

92 definition->freeze();

165std::string TIVIDYFacade::createInstance(const std::string& definitionName) {
166 auto found = definitions.find(definitionName);
167 if (found == definitions.end()) {
168 throw std::invalid_argument("createInstance: unknown definition '" + definitionName + "'");
169 }
170 WorkflowDefinition* definition = found->second;
171
172 std::ostringstream idStream;
173 idStream << "INST-" << ++nextInstanceNumber;
174 const std::string instanceID = idStream.str();
175
176
177 // Prototype: the definition clones its tree into a new, independent instance.
178 WorkflowInstance* instance = definition->createInstance(instanceID);
179 instances[instanceID] = instance;
180
181
182 // Observer: attach every listener so each state change reaches all of them.
183 EventPublisher* publisher = instance->getPublisher();
184 publisher->attach(&audit);
185 publisher->attach(&notifier);
186 publisher->attach(&metrics);
187
188 EngineRouter* router = new EngineRouter(instance, definition->getApprovalChain());
189 routers.push_back(router);
190 publisher->attach(router);
191
192 return instanceID;
193}

254ApprovalStatus TIVIDYFacade::requestApproval(const std::string& instanceID,
255 const std::string& taskID) {
256 WorkflowInstance* instance = findInstance(instanceID);
257 WorkItem* item = findItem(instanceID, taskID);
258
259 ApprovalHandler* chain = instance->getDefinition()->getApprovalChain();
260 if (chain == nullptr) {
261 throw std::runtime_error("requestApproval: the definition has no approval chain");
262 }
263
264 int requiredLevel = DEFAULT_APPROVAL_LEVEL;
265 if (item->hasData(APPROVAL_LEVEL_KEY)) {
266 requiredLevel = std::atoi(item->getData(APPROVAL_LEVEL_KEY).c_str());
267 }
268
269 const std::time_t deadline = std::time(nullptr) + APPROVAL_WINDOW_SECONDS;
270 ApprovalRequest request(item, requiredLevel, deadline);
271
272 // Chain of Responsibility: each level decides or forwards to the next.
273 chain->handle(&request);
274 return request.getStatus();
275}

300WorkItem* TIVIDYFacade::findItem(const std::string& instanceID, const std::string& itemID) {
301 WorkflowInstance* instance = findInstance(instanceID);
302
303 // Iterator: walk the tree without knowing how stages store their children.
304 std::unique_ptr<WorkItemIterator> it(instance->getRoot()->createIterator());
305 for (it->first(); !it->isDone(); it->next()) {
306 WorkItem* item = it->current();
307 if (item->getItemID() == itemID) {
308 return item;
309 }
310 }
311 throw std::invalid_argument("findItem: no item '" + itemID + "' in instance '" + instanceID + "'");
312}
