18TEST_CASE("Director constructs a workflow from a client supplied recipe") {
19 DefinitionBuilder builder;
20 Director director(builder);
21 Role role("Role A", 1);
22 RoleBasedAssignment assignment;
23
24 std::unique_ptr<WorkflowDefinition> definition(director.construct(
25 "Generic workflow",
26 7,
27 [&](WorkflowBuilder& workflow) {
28 workflow.beginSequentialStage("First stage");
29 workflow.addTask("First client-defined task", &role, &assignment);
30 workflow.commitPendingTask();
31 workflow.addTask("Second client-defined task", &role, &assignment);
32 workflow.withPriority(2);
33 workflow.commitPendingTask();
34 workflow.endStage();
35 }
36 ));
37
38 REQUIRE(definition != nullptr);
39 CHECK(definition->getName() == "Generic workflow");
40 CHECK(definition->getVersion() == 7);
41
42 Stage* root = dynamic_cast<Stage*>(definition->getRoot());
43 REQUIRE(root != nullptr);
44 CHECK(root->getChildren().size() == 2);
45}
