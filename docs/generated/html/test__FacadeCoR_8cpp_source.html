21std::string approvalInstance(Fixture& f, const std::string& levelValue, bool setLevel = true) {
22 WorkflowDefinition* def = makeSequentialDefinition("appr", &f.workerRole, 1);
23 if (setLevel) tagTasks(def, "requiredApprovalLevel", levelValue);
24 def->setApprovalChain(makeApprovalChain(&f.nurseRole, &f.doctorRole));
25 f.facade.registerDefinition(def);
26 std::string id = f.facade.createInstance("appr");
27 f.facade.startInstance(id);
28 return id;
29}
