2// Desktop.h - a small header-only Qt window that shows main() as activity
3// diagrams, ONE TAB PER SCENARIO, with a VS Code style terminal under each one:
4//
5// [ Hospital admission ] [ Legal case ] [ Playground ]
6//
7// (●) START ──▶ [ step ] ──▶ [ step ] ──▶ ... ──▶ (◉) END
8// ┌ PROBLEMS OUTPUT DEBUG CONSOLE TERMINAL ───────────────────────────┐
9// │ PS C:\tividy\build> .\workflow_tests.exe --scenario "Legal case" │
10// └───────────────────────────────────────────────────────────────────────┘
11//
12// Every theatre.step(...) is one node, in main's exact words. The running step
13// glows blue, then turns green (passed) or red (failed, with the reason).
14// Anything theatre.say(...) prints appears as a grey note inside the step that
15// said it AND as a line in the terminal. Each tab has its own Run / Step /
16// Restart controls, so scenarios run independently. A tab title gets a tick or
17// a cross once its scenario finishes.
18//
19// API: tab(name) - start a new scenario tab; following step()s belong to it
20// section(name), step(title, detail, body), say(msg), log(msg),
21// check(), expectThrows(), run()
22// No Q_OBJECT / moc needed.

341 void drawNode(QPainter &p, int i, double t) {
342 const TheatreStep &s = (*steps_)[static_cast<std::size_t>(i)];
343 const QRectF r = s.rect;
344 const bool hovered = (hover_ == i);
345 const QColor acc = accent(s.group);
346
347 QColor border("#d3daea"), fill("#ffffff"), badge = acc.lighter(125);
348 QString word = "WAITING";
349 double bw = 1.5;
350 switch (s.state) {
351 case TS_Running: border = QColor("#0a84ff"); fill = QColor("#eef6ff"); badge = border; word = "RUNNING"; bw = 2.5; break;
352 case TS_Passed: border = QColor("#159957"); fill = QColor("#f0faf5"); badge = border; word = "PASSED"; break;
353 case TS_Failed: border = QColor("#d94353"); fill = QColor("#fff3f4"); badge = border; word = "FAILED"; break;
354 default: if (hovered) border = acc.lighter(130); break;
355 }
356
357 // soft drop shadow
358 p.setPen(Qt::NoPen);
359 for (int k = 3; k >= 1; --k) {
360 p.setBrush(QColor(30, 50, 110, 7 + (hovered ? 3 : 0)));
361 p.drawRoundedRect(r.adjusted(-k, 2 - k, k, 4 + k), 14 + k, 14 + k);
362 }
363
364 if (s.state == TS_Running) {
365 const double pulse = 0.5 + 0.5 * std::sin(t * 6.0);
366 for (int k = 1; k <= 3; ++k) {
367 p.setPen(QPen(QColor(10, 132, 255, int((70 - k * 18) * (0.4 + 0.6 * pulse))), 2));
368 p.setBrush(Qt::NoBrush);
369 p.drawRoundedRect(r.adjusted(-k * 2.2, -k * 2.2, k * 2.2, k * 2.2), 14 + k * 2.2, 14 + k * 2.2);
370 }
371 }
372
373 p.setPen(QPen(border, bw));
374 p.setBrush(fill);
375 p.drawRoundedRect(r, 14, 14);
376
377 // coloured edge that matches the section box
378 {
379 QPainterPath clip;
380 clip.addRoundedRect(r, 14, 14);
381 p.save();
382 p.setClipPath(clip);
383 p.setPen(Qt::NoPen);
384 p.setBrush(s.state == TS_Pending ? acc : border);
385 p.drawRect(QRectF(r.left(), r.top(), 5, r.height()));
386 p.restore();
387 }
388
389 // badge: step number, spinner, tick or cross
390 const QPointF c(r.left() + 30, r.top() + std::min(r.height() / 2, 28.0));
391 if (s.state == TS_Running) {
392 p.setPen(QPen(QColor("#0a84ff"), 3, Qt::SolidLine, Qt::RoundCap));
393 p.setBrush(Qt::NoBrush);
394 p.drawArc(QRectF(c.x() - 19, c.y() - 19, 38, 38), int(-t * 540 * 16), 100 * 16);
395 }
396 p.setPen(Qt::NoPen);
397 p.setBrush(badge);
398 p.drawEllipse(c, 15, 15);
399 if (s.state == TS_Passed || s.state == TS_Failed) {
400 p.setPen(QPen(Qt::white, 2.8, Qt::SolidLine, Qt::RoundCap, Qt::RoundJoin));
401 p.setBrush(Qt::NoBrush);
402 if (s.state == TS_Passed) {
403 QPainterPath ck;
404 ck.moveTo(c.x() - 5.5, c.y() + 0.5);
405 ck.lineTo(c.x() - 1.5, c.y() + 4.5);
406 ck.lineTo(c.x() + 6, c.y() - 4.5);
407 p.drawPath(ck);
408 } else {
409 p.drawLine(QPointF(c.x() - 4.5, c.y() - 4.5), QPointF(c.x() + 4.5, c.y() + 4.5));
410 p.drawLine(QPointF(c.x() - 4.5, c.y() + 4.5), QPointF(c.x() + 4.5, c.y() - 4.5));
411 }
412 } else {
413 p.setFont(mk(13, true));
414 p.setPen(Qt::white);
415 p.drawText(QRectF(c.x() - 15, c.y() - 15, 30, 30), Qt::AlignCenter, QString::number(i + 1));
416 }
417
418 // text column
419 const double textX = r.left() + 58;
420 const double titleW = r.width() - 58 - 14 - 76;
421 const double bodyW = r.width() - 58 - 14;
422 double y = r.top() + (r.height() - content(s)) / 2.0;
423
424 p.setFont(mk(14, true));
425 p.setPen(QColor("#1b2333"));
426 p.drawText(QRectF(textX, y, titleW, s.titleH), Qt::TextWordWrap | Qt::AlignLeft | Qt::AlignTop, s.title);
427
428 p.setFont(mk(10, true, 0.8));
429 p.setPen(s.state == TS_Pending ? QColor("#a3adc0") : badge);
430 p.drawText(QRectF(r.right() - 14 - 74, y + 1, 74, 16), Qt::AlignRight | Qt::AlignVCenter, word);
431
432 y += s.titleH;
433 auto para = [&](const QString &str, double h, const QColor &col) {
434 if (h <= 0) return;
435 p.setFont(mk(11));
436 p.setPen(col);
437 p.drawText(QRectF(textX, y, bodyW, h), Qt::TextWordWrap | Qt::AlignLeft | Qt::AlignTop, str);
438 y += h;
439 };
440 if (s.detailH > 0) { y += 3; para(s.detail, s.detailH, QColor("#69758f")); }
441 if (s.noteH > 0) { y += 5; para(s.note, s.noteH, QColor("#8793a5")); }
442 if (s.msgH > 0) { y += 5; para(s.message, s.msgH, QColor("#d94353")); }
443 }
