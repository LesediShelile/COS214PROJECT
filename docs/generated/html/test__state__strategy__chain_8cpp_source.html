239TEST_CASE("Chain of Responsibility: approval levels") {
240 Role nurse("Nurse", 1);
241 Role doctor("Doctor", 2);
242 Role head("Head of Department", 3);
243
244 // Each handler owns its successor, so the levels are created with new
245 // and deleting the first level deletes the whole chain.
246 ApprovalLevel* nurseLevel = new ApprovalLevel(&nurse, 60);
247 ApprovalLevel* doctorLevel = new ApprovalLevel(&doctor, 120);
248 ApprovalLevel* headLevel = new ApprovalLevel(&head, 300);
249 nurseLevel->setNext(doctorLevel);
250 doctorLevel->setNext(headLevel);
251 time_t now = time(nullptr);
252
253 ApprovalRequest low(nullptr, 1, now);
254 nurseLevel->handle(&low);
255 CHECK(low.getStatus() == ApprovalStatus::Approved);
256
257 ApprovalRequest mid(nullptr, 2, now);
258 nurseLevel->handle(&mid);
259 CHECK(mid.getStatus() == ApprovalStatus::Approved);
260 CHECK(mid.getDeadline() >= now + 120);
261
262 ApprovalRequest tooHigh(nullptr, 9, now);
263 nurseLevel->handle(&tooHigh);
264 CHECK(tooHigh.getStatus() == ApprovalStatus::Escalated);
265
266 nurseLevel->handle(&mid);
267 CHECK(mid.getStatus() == ApprovalStatus::Approved);
268
269 CHECK_FALSE(nurseLevel->canApprove(&low));
270 CHECK(doctorLevel->getRole() == &doctor);
271 CHECK(doctorLevel->getTimeout() == 120);
272
273 delete nurseLevel; // also deletes doctorLevel and headLevel
274}
