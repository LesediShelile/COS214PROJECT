2// Playground.h - an interactive workflow sandbox.
3//
4// ┌ header: title · status · Undo · Reset ───────────────────────────────┐
5// │ add bar: [ name... ] [+ Task] [+ Sequential] [+ Parallel] [Templates]│
6// ├──────────────── workflow tree ───────┬──────── inspector ─────────────┤
7// │ stats chips + progress │ selected item, state, owner │
8// │ ▸ Workflow │ actions enabled by state │
9// │ • Task … ● Started │ [Assign as: Alice ▾] │
10// ├──────────────────────────────────────┴────────────────────────────────┤
11// │ terminal: coloured activity log │
12// └───────────────────────────────────────────────────────────────────────┘
13//
14// Ways to play:
15// * build a workflow with the add bar (items go inside the selected stage)
16// * pick a template (Order fulfilment, Hiring pipeline)
17// * select any item and use the inspector buttons, or right-click it
18// * double-click an item to push it forward one state
19// * Step / Auto-play runs the workflow one transition at a time so you can
20// watch the sequential and parallel rules open tasks
21// * Undo walks back assign / start / complete (Command + Memento)
22//
23// Shortcuts: Enter add task · Ctrl+1/2/3 add task/sequential/parallel
24// F6 step · F5 auto-play · Ctrl+Z undo
25// No Q_OBJECT / moc needed.

487 // Where a new item goes: inside the selected stage, next to a selected
488 // task (inside its parent), otherwise under the root.
489 Stage *targetStage() {
490 WorkItem *sel = selected();
491 if (sel) {
492 if (Stage *s = dynamic_cast<Stage*>(sel)) return s;

499 // Create an item of kind 0 = sequential, 1 = parallel, 2 = task, no UI.
500 WorkItem *makeItem(int kind, const QString &name, Stage *parent) {
501 ++nextId_;
502 const std::string id = std::string(kind == 0 ? "seq-" : kind == 1 ? "par-" : "task-")
503 + std::to_string(nextId_);
504 WorkItem *item = nullptr;
505 if (kind == 0) item = new SequentialStage(id, name.toStdString());
506 else if (kind == 1) item = new ParallelStage(id, name.toStdString());
507 else item = new Task(id, name.toStdString(), 0, 0);
508 parent->add(item);
509 names_[item] = name;
510 parents_[item] = parent;
511 return item;
512 }

514 // ======================================================================
515 // Actions (all go through guard so errors land in the log, not a crash)
516 // ======================================================================
517 void guard(const QString &message, const std::function<void()> &action) {
518 try {
519 action();
520 setStatus(message, true);
521 appendLog(Ok, message);
522 } catch (const std::exception &e) {
523 const QString t = QString::fromUtf8(e.what());
524 setStatus(t, false);
525 appendLog(Err, message + " failed: " + t);
526 } catch (...) {
527 setStatus("Unknown exception.", false);
528 appendLog(Err, message + " failed: unknown exception");
529 }
530 releaseNext();
531 refresh();
532 }

606 // Push a task straight through its lifecycle.
607 void driveTask(WorkItem *t) {
608 QString s = stateOf(t);
609 if (s == "Created" || s == "Rejected") { t->makeAvailable(); s = stateOf(t); }
610 if (s == "Available") { runCommand(new AssignCommand(t, currentPerson())); s = stateOf(t); }
611 if (s == "Assigned") { runCommand(new StartCommand(t)); s = stateOf(t); }
612 if (s == "Started") { runCommand(new CompleteCommand(t)); }
613 }
614
615 void driveItem(WorkItem *w) {
616 if (isTask(w)) { driveTask(w); return; }
617 if (stateOf(w) == "Created") w->makeAvailable();
618 std::vector<WorkItem*> tasks;
619 collectTasks(w, tasks);
620 for (int pass = 0; pass < 60; ++pass) {
621 root_->activateChildren();
622 bool any = false;
623 for (std::size_t i = 0; i < tasks.size(); ++i)
624 if (!tasks[i]->isFinished() && releasedItem(tasks[i])) { driveTask(tasks[i]); any = true; }
625 if (!any) break;
626 }
627 }

629 void undo() {
630 if (!invoker_ || !invoker_->canUndo()) {
631 appendLog(Info, "Nothing to undo.");
632 return;
633 }
634 guard("Undid the last action", [this] {

639 // ---- one transition at a time (Step / Auto-play / double-click) ---------
640 // Moves one task a single state forward. Returns a description or an empty
641 // string if nothing could be done.
642 QString transitionOnce(WorkItem *t) {
643 const QString before = stateOf(t);
644 const QString n = displayName(t);
645 if (before == "Created" || before == "Rejected") t->makeAvailable();
646 else if (before == "Available") runCommand(new AssignCommand(t, currentPerson()));
647 else if (before == "Assigned") runCommand(new StartCommand(t));
648 else if (before == "Started") runCommand(new CompleteCommand(t));
649 else return QString();
650 return QString("%1: %2 → %3").arg(n, before, stateOf(t));
651 }
652
653 // Find the first task the workflow has released and advance it one step.
654 bool advanceOnce() {
655 try {
656 if (stateOf(root_) == "Created") {
657 root_->makeAvailable();
658 root_->activateChildren();
659 appendLog(Step, "Workflow opened: first tasks released");
660 refresh();
661 return true;
662 }
663 root_->activateChildren();
664 std::vector<WorkItem*> tasks;
665 collectTasks(root_, tasks);
666 for (std::size_t i = 0; i < tasks.size(); ++i) {
667 WorkItem *t = tasks[i];
668 if (t->isFinished() || !releasedItem(t)) continue;
669 const QString what = transitionOnce(t);
670 if (what.isEmpty()) continue;
671 root_->activateChildren();
672 appendLog(Step, what);
673 setStatus(what, true);
674 refresh();
675 return true;
676 }
677 } catch (const std::exception &e) {
678 appendLog(Err, QString("Step failed: ") + QString::fromUtf8(e.what()));
679 setStatus(QString::fromUtf8(e.what()), false);
680 refresh();
681 return false;
682 }
683 return false;
684 }
